\section{Laboratory Safety Protocols}

\begin{frame}{The Physics of Electric Shock: Human Body as a Resistor}
    \begin{columns}[T]
        \begin{column}{0.52\textwidth}
            \textbf{Ohm's Law Applied to Human Tissue}
            $$I = \frac{V}{R_{\text{body}}}$$
            \begin{itemize}
                \item Voltage alone does not kill; \textbf{current magnitude ($I$)} and pathway dictate lethality.
                \item \textbf{Dry Skin:} $R \approx 10{,}000\,\Omega \text{ to } 100{,}000\,\Omega$.
                \item \textbf{Wet/Perspiring Skin:} $R$ plummets to $\le 1{,}000\,\Omega$.
                \item At $1{,}000\,\Omega$, a $50\,\text{V}$ line yields $50\,\text{mA}$---enough to trigger ventricular fibrillation.
            \end{itemize}
        \end{column}
        \begin{column}{0.44\textwidth}
            \textbf{The ``One-Hand Rule''}
            \begin{itemize}
                \item Keep \textbf{one hand in your pocket} or behind your back when probing live high-voltage test points.
                \item Prevents current from forming a cross-torso circuit through the heart (hand-to-hand pathway).
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Workbench Safety, Housekeeping \& ESD Mitigation}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Personal Protective Equipment (PPE)}
            \begin{itemize}
                \item \textbf{ANSI Z87.1 Safety Glasses:} Mandatory during soldering and component lead clipping.
                \item \textbf{Zero Jewelry Policy:} Rings and metallic watchbands can short power rails, causing severe arc burns.
                \item \textbf{De-energized Prototyping:} Always disconnect power before inserting or pulling components.
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Electrostatic Discharge (ESD) Standards}
            \begin{itemize}
                \item Human perception threshold for ESD is $\approx 3{,}000\,\text{V}$.
                \item CMOS/MOSFET gates suffer punch-through breakdown at $<100\,\text{V}$.
                \item \textbf{Wrist Straps:} Must include a \textbf{$1\,\text{M}\Omega$ safety resistor} to protect user from mains faults.
                \item Handle ICs by packaging body, never exposed pins.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Laboratory Power Tool Safety \& Mechanical Hazards}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{PCB Drills, Rotary Tools \& Heat Guns}
            \begin{itemize}
                \item \textbf{Workpiece Clamping:} Always secure PCBs in a drill press vise or PCB clamp. \textit{Never hand-hold a board while drilling.}
                \item \textbf{Power Disconnection:} Unplug or de-energize the power tool before changing drill bits, collets, or cutting discs.
                \item \textbf{Thermal Awareness:} Heat guns operate above $400^\circ\text{C}$. Rest solely on designated heat-resistant stands.
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Entanglement \& Dust Precautions}
            \begin{itemize}
                \item \textbf{Zero Gloves Policy near Rotating Machinery:} Gloves create a lethal snagging hazard around rotating chucks and drill spindles.
                \item \textbf{Attire Standards:} Tie back long hair, tuck in loose clothing/lanyards, and roll up long sleeves.
                \item \textbf{Respiratory \& Eye Protection:} Wear an N95 dust mask and ANSI Z87.1 eyewear when cutting/drilling FR4 fiberglass to prevent inhaling toxic particulates.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{PCB Etching Chemicals: Safe Storage \& Disposal Protocols}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Safe Chemical Storage ($\text{FeCl}_3$ \& Persulfates)}
            \begin{itemize}
                \item \textbf{Container Compatibility:} Store strictly in heavy-duty, acid-resistant High-Density Polyethylene (HDPE) or polypropylene containers.
                \item \textbf{Zero Metal Contact:} Never use metallic containers or utensils; acidic etchants rapidly corrode metals and release gas.
                \item \textbf{Ventilated Storage:} Keep in dedicated corrosive cabinets away from organic solvents, bases, and heat.
                \item \textbf{Explicit Labeling:} Mark chemical name, concentration, preparation date, and hazard warning.
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Disposal Protocols for Spent Etchant}
            \begin{itemize}
                \item \textbf{Zero Sink Drainage:} \underline{Never} pour spent etchant down domestic or municipal drains. Dissolved copper ($\text{Cu}^{2+}$) is biocidal and corrodes pipes.
                \item \textbf{Neutralization:} Slowly neutralize with sodium bicarbonate ($\text{NaHCO}_3$) until foaming stops.
                \item \textbf{Precipitation \& Collection:} Allow copper hydroxide sludge to settle; collect in labeled hazardous waste carboys.
                \item \textbf{Institutional Transfer:} Transfer waste to University Environmental Safety for certified hazardous disposal.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}
